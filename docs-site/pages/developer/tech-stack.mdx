# 技术栈详解

## Next.js v16 关键差异

### proxy.ts 替代 middleware.ts

```typescript
// src/proxy.ts
export async function proxy(request: Request) {
  // 路由保护逻辑
}
export const config = { matcher: [...] }
```

### 动态路由参数是 Promise

```typescript
// 路由处理
export async function GET(
  request: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  const { id } = await params
  // ...
}
```

### searchParams 也是 Promise

```typescript
// 页面组件
export default async function Page({
  searchParams,
}: {
  searchParams: Promise<{ q?: string }>
}) {
  const { q } = await searchParams
  // ...
}
```

## Prisma v7

### 必须使用 Driver Adapter

```typescript
import { PrismaClient } from "@/generated/prisma/client"
import { PrismaPg } from "@prisma/adapter-pg"

const pool = new Pool({ connectionString: process.env.DATABASE_URL })
const adapter = new PrismaPg(pool)
const prisma = new PrismaClient({ adapter })
```

### 导入路径变更

```typescript
// 客户端
import { PrismaClient } from "@/generated/prisma/client"

// 枚举
import { Role, TemplateStatus } from "@/generated/prisma/enums"
```

### 配置位置

数据库 URL 在 `prisma.config.ts` 中配置，不再使用 schema.prisma 或 .env。

## shadcn/ui v4 (Base UI)

### Button 使用 render prop

```tsx
// 旧版（Radix）
<Button asChild><Link href="..." /></Button>

// 新版（Base UI）
<Button render={<Link href="..." />}>点击</Button>
```

### 使用 sonner 替代 toast

```tsx
import { toast } from "sonner"

toast.success("操作成功")
```

## 认证

### 服务端

```typescript
import { auth } from "@/lib/auth"
const session = await auth()
// session.user.id, session.user.role
```

### 客户端

```typescript
import { useSession, signOut } from "next-auth/react"

const { data: session } = useSession()
```

## 数据库访问

```typescript
import { db } from "@/lib/db"  // 单例，已配置 driver adapter

const templates = await db.template.findMany()
```
